# 接口开发快速参考指南

## 🚀 新接口开发5步骤

### 1️⃣ 创建DTO（5分钟）
```typescript
// dto/create-xxx.dto.ts
import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
import { IsString, IsOptional, MinLength, MaxLength } from 'class-validator';

export class CreateXxxDto {
  @ApiProperty({ description: '名称', example: '示例名称' })
  @IsString({ message: '名称必须是字符串' })
  @MinLength(1, { message: '名称不能为空' })
  @MaxLength(100, { message: '名称不能超过100个字符' })
  name: string;

  @ApiPropertyOptional({ description: '描述' })
  @IsOptional()
  @IsString()
  @MaxLength(500)
  description?: string;
}
```

### 2️⃣ 创建Controller（5分钟）

**普通用户接口**：
```typescript
@ApiTags('功能名称')
@Controller()  // ⚠️ 不要加前缀！
@UseGuards(JwtAuthGuard)  // ⚠️ 必须加认证！
@ApiBearerAuth('JWT-auth')
export class XxxController {
  constructor(private readonly xxxService: XxxService) {}

  @Post()
  @ApiOperation({ summary: '创建资源' })
  @ApiResponse({ status: 201, description: '创建成功' })
  async create(
    @Request() req,
    @Body(ValidationPipe) createDto: CreateXxxDto,
  ) {
    return this.xxxService.create(req.user.id, createDto);
  }
}
```

**管理员接口**：
```typescript
@ApiTags('管理员功能')
@Controller()  // ⚠️ 不要加前缀！
@UseGuards(AdminAuthGuard, RoleGuard)  // ⚠️ 管理员双重守卫！
@ApiBearerAuth('JWT-auth')
export class AdminController {
  @Get('users')
  @Roles('ADMIN')  // ⚠️ 必须加角色！
  @ApiOperation({ summary: '获取用户列表' })
  async getUsers(@Query() query: QueryDto) {
    return this.adminService.getUsers(query);
  }
}
```

### 3️⃣ 创建Service（10分钟）
```typescript
@Injectable()
export class XxxService {
  constructor(private readonly prisma: PrismaService) {}

  async create(userId: string, createDto: CreateXxxDto) {
    return this.prisma.xxx.create({
      data: {
        ...createDto,
        userId,
      },
    });
  }
}
```

### 4️⃣ 注册到Module（2分钟）
```typescript
@Module({
  imports: [DatabaseModule, JwtModule, PassportModule],
  controllers: [XxxController],
  providers: [XxxService, JwtStrategy],
})
export class XxxModule {}
```

### 5️⃣ 在main.ts配置拦截器和过滤器（已配置则跳过）
```typescript
app.useGlobalInterceptors(new ResponseInterceptor());
app.useGlobalFilters(new AllExceptionsFilter());
```

## 📝 常用代码模板

### DTO验证模板

```typescript
// 字符串（必填）
@ApiProperty({ description: '标题' })
@IsString()
@MinLength(1)
@MaxLength(100)
title: string;

// 字符串（可选）
@ApiPropertyOptional({ description: '描述' })
@IsOptional()
@IsString()
@MaxLength(500)
description?: string;

// 邮箱
@ApiProperty({ description: '邮箱', example: 'user@example.com' })
@IsEmail({}, { message: '请输入有效的邮箱' })
@MaxLength(100)
email: string;

// 密码
@ApiProperty({ description: '密码', minLength: 8 })
@IsString()
@MinLength(8, { message: '密码至少8个字符' })
@Matches(/^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)/, {
  message: '密码必须包含大小写字母和数字'
})
password: string;

// 枚举
@ApiProperty({ enum: ['OPTION_A', 'OPTION_B'] })
@IsEnum(['OPTION_A', 'OPTION_B'])
status: string;

// 数字（查询参数）
@ApiPropertyOptional({ description: '页码', default: 1 })
@IsOptional()
@Type(() => Number)
@IsNumber()
page?: number = 1;

// 布尔值
@ApiPropertyOptional({ type: Boolean })
@IsOptional()
@Transform(({ value }) => value === 'true' || value === true)
@IsBoolean()
isActive?: boolean;

// 日期
@ApiPropertyOptional({ description: '开始日期', example: '2024-01-01' })
@IsOptional()
@IsDateString()
startDate?: string;

// 数组
@ApiProperty({ description: '标签', type: [String] })
@IsArray()
@IsString({ each: true })
tags: string[];
```

### Controller方法模板

```typescript
// GET - 列表（分页）
@Get()
@ApiOperation({ summary: '获取列表' })
@ApiQuery({ name: 'page', required: false, type: Number })
@ApiQuery({ name: 'limit', required: false, type: Number })
async findAll(@Request() req, @Query() query: QueryDto) {
  return this.service.findAll(req.user.id, query);
}

// GET - 详情
@Get(':id')
@ApiOperation({ summary: '获取详情' })
@ApiParam({ name: 'id', description: '资源ID' })
@ApiResponse({ status: 404, description: '资源不存在' })
async findOne(@Param('id') id: string, @Request() req) {
  return this.service.findOne(id, req.user.id);
}

// POST - 创建
@Post()
@ApiOperation({ summary: '创建资源' })
@ApiResponse({ status: 201, description: '创建成功' })
@ApiResponse({ status: 400, description: '参数验证失败' })
async create(
  @Request() req,
  @Body(ValidationPipe) createDto: CreateDto,
) {
  return this.service.create(req.user.id, createDto);
}

// PATCH - 更新
@Patch(':id')
@ApiOperation({ summary: '更新资源' })
@ApiParam({ name: 'id', description: '资源ID' })
async update(
  @Param('id') id: string,
  @Request() req,
  @Body(ValidationPipe) updateDto: UpdateDto,
) {
  return this.service.update(id, req.user.id, updateDto);
}

// DELETE - 删除
@Delete(':id')
@ApiOperation({ summary: '删除资源' })
@ApiParam({ name: 'id', description: '资源ID' })
@HttpCode(HttpStatus.OK)  // ⚠️ 必须加！
async remove(@Param('id') id: string, @Request() req) {
  await this.service.remove(id, req.user.id);
  return { message: '删除成功' };
}

// POST - 自定义操作
@Post(':id/action')
@ApiOperation({ summary: '执行操作' })
@ApiParam({ name: 'id', description: '资源ID' })
@HttpCode(HttpStatus.OK)  // ⚠️ 必须加！
async customAction(@Param('id') id: string, @Body() data: any) {
  return this.service.customAction(id, data);
}
```

## ⚠️ 常见错误和解决方案

### 错误1：404 Not Found

| 检查项 | 解决方案 |
|--------|---------|
| ✅ Controller路径 | 微服务使用 `@Controller()` 不加前缀 |
| ✅ API Gateway路由 | 确认Gateway正确代理到微服务 |
| ✅ 服务启动状态 | 确认微服务正常运行 |
| ✅ 端口配置 | 检查.env中的服务端口配置 |

### 错误2：401 Unauthorized

| 检查项 | 解决方案 |
|--------|---------|
| ✅ Guards | 添加 `@UseGuards(JwtAuthGuard)` |
| ✅ ApiBearerAuth | 添加 `@ApiBearerAuth('JWT-auth')` |
| ✅ JwtStrategy | 确认已配置并在Module中提供 |
| ✅ JWT_SECRET | 确认所有服务使用相同的secret |
| ✅ Token | 前端请求头包含 `Authorization: Bearer <token>` |

### 错误3：400 Bad Request

| 检查项 | 解决方案 |
|--------|---------|
| ✅ DTO验证 | 所有字段添加验证装饰器 |
| ✅ ValidationPipe | `@Body(ValidationPipe)` |
| ✅ 类型转换 | 数字用 `@Type(() => Number)` |
| ✅ 布尔转换 | `@Transform(({ value }) => ...)` |
| ✅ 必填字段 | 使用 `@ApiProperty` 不是 `@ApiPropertyOptional` |

### 错误4：403 Forbidden（管理员接口）

| 检查项 | 解决方案 |
|--------|---------|
| ✅ AdminAuthGuard | 添加并配置AdminAuthGuard |
| ✅ RoleGuard | 添加RoleGuard到@UseGuards |
| ✅ @Roles装饰器 | 方法上添加 `@Roles('ADMIN')` |
| ✅ 用户角色 | 确认JWT payload中role为'ADMIN' |

## 🎯 开发前检查清单

### 开始编码前（必须完成）

- [ ] 确认接口属于哪个微服务（admin/user/novel/ai/payment）
- [ ] 确认是普通用户接口还是管理员接口
- [ ] 确认需要哪些DTO（Create/Update/Query）
- [ ] 确认是否需要权限验证

### 编码完成后（必须检查）

- [ ] Controller不使用路径前缀（微服务中）
- [ ] 添加了认证Guards（JwtAuthGuard或AdminAuthGuard+RoleGuard）
- [ ] 管理员接口添加了@Roles装饰器
- [ ] DTO所有字段都有验证装饰器
- [ ] 所有方法添加了@ApiOperation
- [ ] 所有方法添加了@ApiResponse
- [ ] DELETE/POST非创建操作添加了@HttpCode(HttpStatus.OK)
- [ ] 从req.user获取用户信息而不是从参数

### 测试前（必须验证）

- [ ] 服务能正常启动
- [ ] API Gateway能访问微服务
- [ ] Swagger文档正常显示
- [ ] 测试无token访问返回401
- [ ] 测试有效token能正常访问
- [ ] 测试参数验证正常工作

## 📚 重要提醒

### ⚠️ 绝对不要做

1. ❌ 微服务Controller添加路径前缀（如 `@Controller('admin')`）
2. ❌ 忘记添加Guards（所有接口都需要认证）
3. ❌ DTO缺少验证装饰器
4. ❌ 直接从参数获取userId（应该从req.user获取）
5. ❌ 忘记@HttpCode导致DELETE返回204
6. ❌ 管理员接口忘记@Roles装饰器

### ✅ 必须要做

1. ✅ 微服务Controller使用 `@Controller()`
2. ✅ 所有接口添加Guards和ApiBearerAuth
3. ✅ DTO完整的验证装饰器
4. ✅ 从req.user.id获取当前用户ID
5. ✅ DELETE/POST非创建操作添加@HttpCode(HttpStatus.OK)
6. ✅ 所有方法添加Swagger文档注解

## 🚀 快速开始模板

**复制这个模板开始新接口开发**：

```typescript
// ========== DTO ==========
import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
import { IsString, IsOptional, MinLength, MaxLength } from 'class-validator';

export class CreateXxxDto {
  @ApiProperty({ description: '名称' })
  @IsString()
  @MinLength(1)
  @MaxLength(100)
  name: string;
}

export class UpdateXxxDto {
  @ApiPropertyOptional({ description: '名称' })
  @IsOptional()
  @IsString()
  @MinLength(1)
  @MaxLength(100)
  name?: string;
}

export class QueryXxxDto {
  @ApiPropertyOptional({ description: '页码', default: 1 })
  @IsOptional()
  @Type(() => Number)
  @IsNumber()
  page?: number = 1;

  @ApiPropertyOptional({ description: '每页数量', default: 20 })
  @IsOptional()
  @Type(() => Number)
  @IsNumber()
  limit?: number = 20;
}

// ========== Controller ==========
import {
  Controller, Get, Post, Patch, Delete,
  Body, Param, Query, UseGuards, Request,
  ValidationPipe, HttpCode, HttpStatus,
} from '@nestjs/common';
import {
  ApiTags, ApiOperation, ApiResponse,
  ApiBearerAuth, ApiParam, ApiQuery,
} from '@nestjs/swagger';
import { JwtAuthGuard } from '@app/common';

@ApiTags('功能名称')
@Controller()
@UseGuards(JwtAuthGuard)
@ApiBearerAuth('JWT-auth')
export class XxxController {
  constructor(private readonly xxxService: XxxService) {}

  @Post()
  @ApiOperation({ summary: '创建资源' })
  @ApiResponse({ status: 201, description: '创建成功' })
  async create(
    @Request() req,
    @Body(ValidationPipe) createDto: CreateXxxDto,
  ) {
    return this.xxxService.create(req.user.id, createDto);
  }

  @Get()
  @ApiOperation({ summary: '获取列表' })
  @ApiQuery({ name: 'page', required: false, type: Number })
  @ApiQuery({ name: 'limit', required: false, type: Number })
  async findAll(@Request() req, @Query() query: QueryXxxDto) {
    return this.xxxService.findAll(req.user.id, query);
  }

  @Get(':id')
  @ApiOperation({ summary: '获取详情' })
  @ApiParam({ name: 'id', description: '资源ID' })
  async findOne(@Param('id') id: string, @Request() req) {
    return this.xxxService.findOne(id, req.user.id);
  }

  @Patch(':id')
  @ApiOperation({ summary: '更新资源' })
  @ApiParam({ name: 'id', description: '资源ID' })
  async update(
    @Param('id') id: string,
    @Request() req,
    @Body(ValidationPipe) updateDto: UpdateXxxDto,
  ) {
    return this.xxxService.update(id, req.user.id, updateDto);
  }

  @Delete(':id')
  @ApiOperation({ summary: '删除资源' })
  @ApiParam({ name: 'id', description: '资源ID' })
  @HttpCode(HttpStatus.OK)
  async remove(@Param('id') id: string, @Request() req) {
    await this.xxxService.remove(id, req.user.id);
    return { message: '删除成功' };
  }
}

// ========== Service ==========
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '@app/database';

@Injectable()
export class XxxService {
  constructor(private readonly prisma: PrismaService) {}

  async create(userId: string, createDto: CreateXxxDto) {
    return this.prisma.xxx.create({
      data: { ...createDto, userId },
    });
  }

  async findAll(userId: string, query: QueryXxxDto) {
    const { page = 1, limit = 20 } = query;
    const skip = (page - 1) * limit;

    const [items, total] = await Promise.all([
      this.prisma.xxx.findMany({
        where: { userId },
        skip,
        take: limit,
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.xxx.count({ where: { userId } }),
    ]);

    return {
      items,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    };
  }

  async findOne(id: string, userId: string) {
    const item = await this.prisma.xxx.findFirst({
      where: { id, userId },
    });

    if (!item) {
      throw new NotFoundException('资源不存在');
    }

    return item;
  }

  async update(id: string, userId: string, updateDto: UpdateXxxDto) {
    await this.findOne(id, userId);
    
    return this.prisma.xxx.update({
      where: { id },
      data: updateDto,
    });
  }

  async remove(id: string, userId: string) {
    await this.findOne(id, userId);
    
    await this.prisma.xxx.delete({
      where: { id },
    });
  }
}

// ========== Module ==========
import { Module } from '@nestjs/common';
import { JwtModule } from '@nestjs/jwt';
import { PassportModule } from '@nestjs/passport';
import { DatabaseModule } from '@app/database';
import { XxxController } from './xxx.controller';
import { XxxService } from './xxx.service';
import { JwtStrategy } from '../../strategies/jwt.strategy';

@Module({
  imports: [
    DatabaseModule,
    PassportModule,
    JwtModule,
  ],
  controllers: [XxxController],
  providers: [XxxService, JwtStrategy],
  exports: [XxxService],
})
export class XxxModule {}
```
