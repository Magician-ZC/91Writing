### 2. 乐观并发控制

```typescript
// 章节更新冲突检测
if (chapter.updatedAt > dto.lastKnownUpdatedAt) {
  throw new ConflictException({
    serverVersion: chapter,
    clientVersion: dto,
    strategies: ['keep-server', 'keep-client', 'merge']
  });
}
```

### 3. 实时配额管理

```typescript
// 自动同步真实使用量
const actualUsedSpace = await this.prisma.material.aggregate({
  _sum: { fileSize: true }
});

if (quota.usedSpace !== actualUsedSpace) {
  // 自动修正不一致
  await this.updateQuota(actualUsedSpace);
}
```
